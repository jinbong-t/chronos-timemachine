function startDrift() {
  const box = $('#drift'); if (!box) return; const id = ++driftRun; const W = box.clientWidth, H = box.clientHeight;
  const chips = [...box.children].map(el => { const w = el.offsetWidth, h = el.offsetHeight, sp = 55 + Math.random()*60, a = Math.random()*Math.PI*2; return {el, w, h, x:Math.random()*Math.max(1,W-w), y:Math.random()*Math.max(1,H-h), vx:Math.cos(a)*sp, vy:Math.sin(a)*sp}; });
  let last = performance.now();
  (function f(t) {
    if (id !== driftRun || !document.contains(box)) return; const dt = Math.min(.05, (t-last)/1000); last = t; const k = reduced ? 0 : (St.slow ? .25 : 1);
    chips.forEach(c => { c.x += c.vx*dt*k; c.y += c.vy*dt*k; if (c.x < 0 || c.x > W-c.w) { c.vx *= -1; c.x = Math.max(0, Math.min(W-c.w, c.x)); } if (c.y < 0 || c.y > H-c.h) { c.vy *= -1; c.y = Math.max(0, Math.min(H-c.h, c.y)); }
      if (k > 0 && Math.random() < .0035*k) { c.x = Math.random()*Math.max(1,W-c.w); c.y = Math.random()*Math.max(1,H-c.h); c.el.classList.remove('jump'); void c.el.offsetWidth; c.el.classList.add('jump'); }
      c.el.style.transform = `translate(${c.x}px,${c.y}px)`; });
    requestAnimationFrame(f);
  })(last);
}
