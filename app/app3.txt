const JOSA = {'이가':['이','가','이(가)'],'을를':['을','를','을(를)'],'은는':['은','는','은(는)']};
function josa(word, key) { const j = JOSA[key]; if (!word) return j[2]; const c = word.charCodeAt(word.length-1) - 0xAC00; if (c < 0 || c > 11171) return j[2]; return c % 28 ? j[0] : j[1]; }
function partText(parts, b, idx) { const p = parts[idx]; if (typeof p !== 'string' || p[0] !== '#') return p; const prev = parts[idx-1]; return josa(typeof prev === 'number' ? b[prev] : '', p.slice(1,3)) + p.slice(3); }
function blanksOf(i) { const T = St.T; const n = WB[i].parts.filter(p => typeof p === 'number').length; if (!T.blanks[i] || T.blanks[i].length !== n) T.blanks[i] = Array(n).fill(''); return T.blanks[i]; }
function sentenceHTML(i, prefix) {
  const b = blanksOf(i);
  return `<div class="sentence">${prefix || ''}${WB[i].parts.map((p,idx) => typeof p === 'number' ? `<button class="blank ${b[p]?'':'empty'} ${St.blankSel===p?'sel':''}" data-act="blank" data-b="${p}">${b[p] ? esc(b[p]) : '빈칸 ' + (p+1)}</button>` : esc(partText(WB[i].parts, b, idx))).join('')}</div>`;
}
