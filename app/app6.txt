  if (a === 'ph-credits') return Store.patchSession(Tc.cls, {phase:'credits'});
  if (a === 'warp') return Store.saveTeam(Tc.cls, b.dataset.t, {warpReq: Tc.teams[b.dataset.t].scene});
  if (a === 'star') { const t = b.dataset.t, k = b.dataset.k, T = Tc.teams[t]; const f = (Tc.sess.featured || []).slice(); const i = f.findIndex(x => x.tid === t && x.k === k);
    if (i >= 0) f.splice(i,1); else f.push({tid:t, k, team:T.name, text: k === 'r' ? T.reflection : (k === '3' ? dilSentence(T) : T.reasons[k])}); Store.patchSession(Tc.cls, {featured:f}); return; }
  if (a === 'copyTsv') { const rows = [['모둠','S#1 기술의 의미','S#2 발달과 사회 변화','S#3 윤리 딜레마','S#4 표준화','S#5 적정기술','성찰','힌트권 사용','귀환']];
    teamList().forEach(t => { const T = Tc.teams[t]; rows.push([T.name, T.reasons[1]||'', T.reasons[2]||'', dilSentence(T), T.reasons[4]||'', T.reasons[5]||'', T.reflection||'', 2-(T.hintsLeft==null?2:T.hintsLeft), T.finished?'O':'']); });
    const tsv = rows.map(r => r.map(c => String(c).replace(/[\t\n]/g,' ')).join('\t')).join('\n');
    const fb = () => { const ta = document.createElement('textarea'); ta.value = tsv; ta.rows = 6; b.parentElement.after(ta); ta.select(); $('#copymsg').textContent = '복사가 막혀 있어요. 아래 글을 선택해 복사하세요.'; };
    try { navigator.clipboard.writeText(tsv).then(() => { $('#copymsg').textContent = '복사했어요. 구글 시트에 붙여 넣으세요.'; }, fb); } catch(err) { fb(); } return; }
  if (a === 'jump') { const t = $('#jumpTeam').value, n = +b.dataset.n; Store.saveTeam(Tc.cls, t, {jumpReq:{scene:n, at:now()}}); if (Tc.sess.phase !== 'play') Store.patchSession(Tc.cls, {phase:'play', startedAt: Tc.sess.startedAt || now()}); return; }
  if (a === 'setCls') { const v = $('#clsIn').value.trim(); if (v) Store.setClass(v); return; }
  if (a === 'reset') { $('#resetConfirm').innerHTML = `<span class="small">모든 모둠 기록이 지워집니다.</span> <button class="btn small" style="--c:var(--red)" data-act="resetYes">지우기</button>`; return; }
  if (a === 'resetYes') { Store.resetClass(Tc.cls); $('#resetConfirm').textContent = '초기화했습니다.'; return; }
}
